107 /*
108  Calculate the crctab once on init.
109 
110  The generating polynomial is
111 
112  32 26 23 22 16 12 11 10 8 7 5 4 2 1
113  G(X)=X + X + X + X + X + X + X + X + X + X + X + X + X + X + 1
114 
115  The i bit in GEN is set if X^i is a summand of G(X) except X^32.
116 
117  #define GEN (BIT (26) | BIT (23) | BIT (22) | BIT (16) | BIT (12) \
118  | BIT (11) | BIT (10) | BIT (8) | BIT (7) | BIT (5) \
119  | BIT (4) | BIT (2) | BIT (1) | BIT (0))
120 
121  which produces 0x04c11db7
122 */

146 static uint32_t crc_remainder(int m, uint32_t *r)
147 {
148  uint32_t rem = 0;
149  int i;
150 
151  for (i = 0; i < 8; i++) {
152  if (BIT (i) & m) {
153  rem ^= r[i];
154  }
155  }
156 
157  return rem & 0xFFFFFFFF; /* Make it run on 64-bit machine */
158 }

167 static int init_crctab(void)
168 {
169  int i;
170  uint32_t r[8];
171 
172  /* Fill r */
173  r[0] = GEN;
174  for (i = 1; i < 8; i++) {
175  r[i] = (r[i - 1] << 1) ^ ((r[i - 1] & SBIT) ? GEN : 0);
176  }
177 
178  crctab[0] = 0x00000000;
179  for (i = 1; i < 256; i++) {
180  crctab[i] = crc_remainder(i, r);
181  }
182 
183  return 1;
184 }

201 unsigned int cksum(unsigned char const *buf, size_t length)
202 {
203  uint32_t crc = 0;
204  size_t bytes = length;
205 
206 #ifndef STATIC_CRCTAB
207  static short inited = 0;
208  if (!inited) inited = init_crctab();
209 #endif
210 
211  while (bytes--) {
212  crc = (crc << 8) ^ crctab[((crc >> 24) ^ *buf++) & 0xFF];
213  }
214 
215  for (; length; length >>= 8) {
216  crc = (crc << 8) ^ crctab[((crc >> 24) ^ length) & 0xFF];
217  }
218 
219  crc = ~crc & 0xFFFFFFFF;
220 
221  return crc;
222 }
