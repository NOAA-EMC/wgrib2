28 ! grb2_mk_inv('FILE.grb', 'FILE.inv') same as wgrib2 FILE.grb -Match_inv >FILE.inv

30 ! The inventory file can be a memory file if has the name @mem:N N=0..29

33 ! grb2_filter('IN.grb', 'OUT.grb', ...)

38 ! grb2_inq :: grib2 inquire
39 ! grb2_inq('IN.grb', 'IN.inv', ...)
40 ! uses memory files @mem:19 .. always

42 ! grb2_wrt :: grib2 write
43 ! grb2_wrt('OUT.grb', 'TEMPLATE.grb', 'ID', ...)
44 ! TEMPLATE.grb is a grib file that is used as a template for the new grib file
