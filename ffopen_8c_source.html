230 int ffclose(FILE *flocal)
231 {
232  struct opened_file *ptr, *last_ptr;
233 
234  if (flocal == stdin || flocal == stdout) return 0; // do not close stdin or stdout
235 
236  last_ptr = NULL;
237  ptr = opened_file_start;
238  while (ptr != NULL) {
239  if (ptr->handle == flocal) {
240  if (ptr->usage_count <= 0) fatal_error_i("ffclose: usage_count = %d <= 0, %d", ptr->usage_count);
241  if (--ptr->usage_count == 0) { // usage_count is zero, can close file name
242  if (ptr->do_not_close_flag == 1) return 0;
243  if (last_ptr == NULL)
244  opened_file_start = ptr->next;
245  else {
246  last_ptr->next = ptr->next;
247  }
248  fclose(ptr->handle);
249  free(ptr->name);
250  free(ptr);
251  }
252  return 0;
253  }
254  last_ptr = ptr;
255  ptr = ptr->next;
256  }
257  return 1; /* not found */
258 }
