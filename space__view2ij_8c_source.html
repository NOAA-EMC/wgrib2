222 
223  if (plat > 90.0 || plat < -90.0) return -1;
224 
225  plon *= (M_PI/180.0);
226  plat *= (M_PI/180.0);
227 
228  /* preliminary check to see if point is visable */
229  if (fmod( fabs(plat-lap), M_PI) > M_PI_2) return -1;
230 
231 // printf(" calc clat/phi_e r_pol_eq = %lf\n",r_pol_eq);
232  phi_e = atan(r_pol_eq*tan(plat));
233  cos_phi_e = cos(phi_e);
234 
235  r_e = r_pol / sqrt(1.0 - factor_10 * cos_phi_e*cos_phi_e);
236 // printf("r_e %lf factor_10 %lf sat_height %lf\n", r_e, factor_10, sat_height);
237 // printf("r_eq %lf r_pol %lf xp %lf yp %lf\n", r_eq, r_pol, xp, yp);
238 
239  r1 = sat_height - r_e*cos_phi_e*cos(plon - lop);
240  r2 = -r_e * cos_phi_e * sin(plon - lop);
241  r3 = r_e * sin(phi_e);
242  rn = sqrt(r1*r1 + r2*r2 + r3*r3);
243 
244 // printf("r1 %lg r2 %lg r3 %lg rn %lg\n", r1, r2, r3, rn);
245 
246  // test to see if the point is visible
247  // barely visible if the triangle between the center of the earth
248  // the satellite and the point on the earth is a right angle triangle

250

252  if (sat_height*sat_height <= rn*rn + r_e*r_e) return -1;
253

255  y = asin(r3/rn);

258  y *= inv_ry;
