143  if (basic_ang != 0) {

155  lon2 = GDS_LatLon_lon2(gds) * units;

157  if (lon1 < 0.0 || lon2 < 0.0) fatal_error("BAD grid definition lon < zero","");
158  if (lon1 > 360.0 || lon2 > 360.0) fatal_error("BAD grid definition lon >= 360","");
159  if (lat1 < -90.0 || lat2 < -90.0 || lat1 > 90.0 || lat2 > 90.0) fatal_error("BAD grid definition lat","");

164  n = lat2;
165  }

167  s = lat2;
168  n = lat1;
169  }

171 
172  if (nny > 1) {
173  dy = (n - s) / (nny - 1);
174  if (nres & 16) { /* lat increment is valid */
175  if (fabs(dy - dlat) > 0.001) fatal_error("lat-lon grid: dlat is inconsistent, bad grid definition","");
176  }
177  }
178  else {
179  dy = 0.0;
180  }

182 
183  /* find W latitude and dx */
184 
185  if ( GDS_Scan_row_rev(nscan) && (nny % 2 == 0) && ((nres & 32) == 0) ) {
186  fatal_error("grib grid definition ambiguity","");
187  }
188

190  w = lon1;
191  e = lon2;
192  if (GDS_Scan_row_rev(nscan) && ((nres & 32) == 0)) {
193  e = lon1 + (nnx-1) * dlon;
194  }
195  } else {
196  w = lon2;
197  e = lon1;
198  if (GDS_Scan_row_rev(nscan) && ((nres & 32) == 0)) {
199  w = lon1 - (nnx-1) * dlon;
200  }
201  }
202 
203  if (e <= w) e += 360.0;
204  if (e-w > 360.0) e -= 360.0;
205  if (w < 0) {
206  w += 360.0;
207  e += 360.0;
208  }
209 
210  /* lat-lon should be in a WE:SN order */
211 
212  if (nnx > 0 && nny > 0) { /* non-thinned, potentially staggered grid */
213  /* put x[] and y[] values in lon[] and lat[] */
214  llat = *lat;
215  llon = *lon;
216  if (stagger(sec, nnpnts,llon,llat)) fatal_error("geo: stagger problem","");
217 
218  if (nnx > 1) {
219  dx = (e-w) / (nnx - 1);
220  dx = fabs(dx);
221  if (nres & 32) { /* lon increment is valid */
222  if (fabs(dx - fabs(dlon)) > 0.001) fatal_error("lat-lon grid: dlon is inconsistent, bad grid definition","");
223  }
224  }
225  else {
226  dx = 0.0;
227  }
228  dy = fabs(dy);
229 
230 #ifdef USE_OPENMP
231 #pragma omp parallel for private(j)
232 #endif
233  for (j = 0; j < nnpnts; j++) {
234  llon[j] = lon1 + llon[j]*dx;
235  llon[j] = llon[j] >= 360.0 ? llon[j] - 360.0 : llon[j];
236  llon[j] = llon[j] < 0.0 ? llon[j] + 360.0 : llon[j];
237  llat[j] = lat1 + llat[j]*dy;
238  }
239  return 0;
240  }
241 
242  /* must be thinned grid */
243 
244  llat = *lat;
245  /* quasi-regular grid */
246  for (j = 0; j < nny; j++) {
247  for (i = 0; i < variable_dim[j]; i++) {
248  *llat++ = s + j*dy;
249  }
250  }
251 
252  llon = *lon;
253  /* quasi-regular grid */
254  for (j = 0; j < nny; j++) {
255  dx = (e-w) / (variable_dim[j]-1);
256  for (i = 0; i < variable_dim[j]; i++) {
257  *llon++ = w + i*dx >= 360.0 ? w + i*dx - 360.0: w + i*dx;
258  }
259  }
260  return 0;
261 } /* end regular2ll() */

368  int nres, nscan;

370  size_t nx;

372  get_nxny_(sec, &nnx, &nny, &nnpnts, &nres, &nscan);

374  gds = sec[3];
375 
376  if (nnx < 1 || nny < 1) {
377  fprintf(stderr,"Sorry code does not handle variable nx/ny yet\n");
378  return 0;
379  }
380  nx = nnx; /* size_t, multiplications will not overflow */
381 
382  if ((*llat = (double *) malloc(((size_t) nnpnts) * sizeof(double))) == NULL) {
383  fatal_error("polar2ll memory allocation failed","");
384  }
385  if ((*llon = (double *) malloc(((size_t) nnpnts) * sizeof(double))) == NULL) {
386  fatal_error("polar2ll memory allocation failed","");
387  }

389  lon = *llon;
390 
391 
392  /* based on iplib */
393 
394  lat1 = GDS_Polar_lat1(gds) * (M_PI/180);
395  lon1 = GDS_Polar_lon1(gds);
396  orient = GDS_Polar_lov(gds);
397  LatD = GDS_Polar_lad(gds);
398 
399  lon1 *= (M_PI/180.0);
400  orient *= (M_PI/180.0);
401 
402  dy = GDS_Polar_dy(gds);
403  dx = GDS_Polar_dx(gds);
404 
405  h = 1.0;
406  if (GDS_Polar_sps(gds)) {
407  h = -1.0;
408  /* added 12/19/2008 WNE sps checkout */
409  orient -= M_PI;
410  }
411 
412 // removed 12/11 if (! (GDS_Scan_x(nscan))) dx = -dx;
413 // removed 12/11 if (! (GDS_Scan_y(nscan))) dy = -dy;
414 
415  /* 60 probably becomes something else in grib2 */
416  /* vsm: from comment to grib2 polar template:
417  "Grid length is in units of 10-3 m at the latitude specified by LaD"
418  do use GDS_Polar_lad(gds) instead of 60?
419  Do use fabs for southern hemisphere?
420  */
421 
422  de = (1.0 + sin(fabs(LatD)*(M_PI/180.0))) * radius_earth(sec);
423  dr = de * cos(lat1) / (1 + h*sin(lat1));
424 
425  xp=-h*sin(lon1-orient)*dr/dx;
426  yp= cos(lon1-orient)*dr/dy;
427 
428 // added 12/11
429  if (! (GDS_Scan_y(nscan))) {
430  yp = yp - nny + 1;
431  }
432  if (! (GDS_Scan_x(nscan))) {
433  xp = xp - nnx + 1;
434  }
435 
436  de2 = de*de;
437 #ifdef USE_OPENMP
438 #pragma omp parallel for private(iy,ix,di,dj,dr2,tmp)
439 #endif
440  for (iy = 0; iy < nny; iy++) {
441  for (ix = 0; ix < nnx; ix++) {
442  di = (ix - xp) * dx;
443  dj = (iy - yp) * dy;
444  dr2 = di*di + dj*dj;
445  if (dr2 < de2*1e-6) {
446  lon[ix+iy*nx] = 0.0;
447  lat[ix+iy*nx] = h*90.0;
448  } else {
449  tmp = (orient+h*atan2(di,-dj))*(180.0/M_PI);
450  if (tmp < 0.0) tmp += 360.0;
451  if (tmp > 360.0) tmp -= 360.0;
452  lon[ix+iy*nx] = tmp;
453  lat[ix+iy*nx] = h*asin((de2-dr2)/(de2+dr2))*(180.0/M_PI);
454  }
455  }
456  }
457  return 0;
458 }

471 int lambert2ll(unsigned char **sec, double **llat, double **llon) {
472 
473  double n;
474  double *lat, *lon;
475 
476  double dx, dy, lat1r, lon1r, lon2d, lon2r, latin1r, latin2r;
477  double lond, latd, d_lon;
478  double f, rho, rhoref, theta, startx, starty;
479  unsigned int nnx, nny;
480  int nres, nscan;
481  double x, y, tmp;
482  unsigned char *gds;
483  double latDr;
484  double earth_radius;
485  unsigned int j, nnpnts;
486 
487  get_nxny_(sec, &nnx, &nny, &nnpnts, &nres, &nscan);
488 
489  if (nnx < 1 || nny < 1) {
490  fprintf(stderr,"Sorry code does not handle variable nx/ny yet\n");
491  return 0;
492  }
493 
494  earth_radius = radius_earth(sec);
495  gds = sec[3];
496  dy = GDS_Lambert_dy(gds);
497  dx = GDS_Lambert_dx(gds);
498  lat1r = GDS_Lambert_La1(gds) * (M_PI / 180.0);
499  lon1r = GDS_Lambert_Lo1(gds) * (M_PI / 180.0);
500  lon2d = GDS_Lambert_Lov(gds);
501  lon2r = lon2d * (M_PI / 180.0);
502  latin1r = GDS_Lambert_Latin1(gds) * (M_PI/180.0);
503  latin2r = GDS_Lambert_Latin2(gds) * (M_PI/180.0);
504 
505 // fix for theta start value crossing 0 longitude
506 // if ((lon1r - lon2r) > 0) lon2r = lon2r + 2*M_PI;
507 
508 
509 //
510 // Latitude of "false origin" where scales are defined.
511 // It is used to estimate "reference_R", rhoref.
512 // Often latDr == latin1r == latin2r and non-modified code is true and works fine.
513 // But could be different if intersection latitudes latin1r and latin2r are different.
514 // Usually latDr must be latin1r <= latDr <= latin2r, other could be strange.
515 //
516  latDr = GDS_Lambert_LatD(gds) * (M_PI/180.0);
517 
518 
519  if (lon1r < 0) fatal_error("bad grid definition, lon1r < 0.0","");
520 
521  if ( fabs(latin1r - latin2r) < 1E-09 ) {
522  n = sin(latin1r);
523  }
524  else {
525  n = log(cos(latin1r)/cos(latin2r)) /
526  log(tan(M_PI_4 + latin2r/2.0) / tan(M_PI_4 + latin1r/2.0));
527  }
528 
529  f = (cos(latin1r) * pow(tan(M_PI_4 + latin1r/2.0), n)) / n;
530 
531  rho = earth_radius * f * pow(tan(M_PI_4 + lat1r/2.0),-n);
532  // old rhoref = earth_radius * f * pow(tan(M_PI_4 + latin1r/2.0),-n);
533  rhoref = earth_radius * f * pow(tan(M_PI_4 + latDr/2.0),-n);
534 
535  // 2/2009 .. new code
536  d_lon = lon1r - lon2r;
537  if (d_lon > M_PI) d_lon -= 2*M_PI;
538  if (d_lon < -M_PI) d_lon += 2*M_PI;
539  theta = n * d_lon;
540  // 2/2009 theta = n * (lon1r - lon2r);
541 
542  startx = rho * sin(theta);
543  starty = rhoref - rho * cos(theta);
544 
545  if ((*llat = (double *) malloc(((size_t) nnpnts) * sizeof(double))) == NULL) {
546  fatal_error("lambert2ll memory allocation failed","");
547  }
548  if ((*llon = (double *) malloc(((size_t) nnpnts) * sizeof(double))) == NULL) {
549  fatal_error("lambert2ll memory allocation failed","");
550  }
551  lat = *llat;
552  lon = *llon;
553 
554  /* put x[] and y[] values in lon[] and lat[] */
555  if (stagger(sec, nnpnts, lon, lat)) fatal_error("geo: stagger problem","");
556 
557  dx = fabs(dx);
558  dy = fabs(dy);
559 
560 #ifdef USE_OPENMP
561 #pragma omp parallel for private(j,x,y,tmp,theta,rho,lond,latd)
562 #endif
563  for (j = 0; j < nnpnts; j++) {
564  y = starty + lat[j]*dy;
565  x = startx + lon[j]*dx;
566  tmp = rhoref - y;
567  theta = atan(x / tmp);
568  rho = sqrt(x * x + tmp*tmp);
569  rho = n > 0 ? rho : -rho;
570  lond = lon2d + todegrees(theta/n);
571  latd = todegrees(2.0 * atan(pow(earth_radius * f/rho,1.0/n)) - M_PI_2);
572  lond = lond >= 360.0 ? lond - 360.0 : lond;
573  lond = lond < 0.0 ? lond + 360.0 : lond;
574  lon[j] = lond;
575  lat[j] = latd;
576  }
577  return 0;
578 } /* end lambert2ll() */

774 double *gauss2lats(int nlat, double *ylat) {
775 
776  const double xlim = 1.0E-7;
777 
778  double *cosc = (double *) malloc(sizeof(double) * (nlat + 1));
779  double *sinc = (double *) malloc(sizeof(double) * (nlat + 1));
780  double *colat = (double *) malloc(sizeof(double) * (nlat + 1));
781 
782  int nzero = (nlat / 2);
783 
784  int i;
785  double fi = nlat;
786  double fi1 = fi + 1.0;
787  double a = fi * fi1/sqrt(4.0*fi1*fi1 - 1.0);
788  double b = fi1 * fi/sqrt(4.0*fi*fi - 1.0);
789 
790  double g, gm, gp, gt, delta, d;
791 
792 #ifdef USE_OPENMP
793 #pragma omp parallel for private(i,g,gm,gp,gt,delta)
794 #endif
795  for (i = 1; i <= nzero; i++) {
796  cosc[i] = sin((i - 0.5)*M_PI/nlat + M_PI*0.5);
797 
798  g = gord(nlat, cosc[i]);
799  gm = gord(nlat - 1, cosc[i]);
800  gp = gord(nlat + 1, cosc[i]);
801  gt = (cosc[i]*cosc[i] - 1.0)/(a * gp - b * gm);
802  delta = g*gt;
803  cosc[i] = cosc[i] - delta;
804 
805  while ( fabs(delta) > xlim ) {
806  g = gord(nlat,cosc[i]);
807  gm = gord(nlat - 1, cosc[i]);
808  gp = gord(nlat + 1, cosc[i]);
809  gt = (cosc[i]*cosc[i] - 1.0)/(a * gp - b * gm);
810  delta = g*gt;
811  cosc[i] = cosc[i] - delta;
812 
813  } /* end while */
814 
815  colat[i] = acos(cosc[i]);
816  sinc[i] = sin(colat[i]);
817  }
818 
819  /*
820  * ... deal with equator if odd number of points
821  */
822  if ( ( nlat % 2) != 0 ) {
823  i = nzero + 1;
824  cosc[i] = 0.0;
825  d = gord(nlat - 1, cosc[i]);
826  d = d*d*fi*fi;
827  colat[i] = M_PI * 0.5;
828  sinc[i] = 1.0;
829  } /* end if() */
830 
831  /*
832  * ... deal with southern hemisphere by symmetry
833  */
834  for (i = nlat - nzero + 1; i <= nlat; i++) {
835  cosc[i] = -cosc[nlat + 1 - i];
836  colat[i] = M_PI - colat[nlat + 1 - i];
837  sinc[i] = sinc[nlat + 1 - i];
838  } /* end for(i) */
839 
840 #ifdef USE_OPENMP
841 #pragma omp parallel for private(i)
842 #endif
843  for (i = 1; i <= nlat; i++) {
844  ylat[i-1] = todegrees(acos(sinc[i]));
845  if ( i > (nlat / 2) ) ylat[i-1] = -ylat[i-1];
846  /* change from N-S to S-N */
847  ylat[i-1] = -ylat[i-1];
848  }
849 
850  free(cosc);
851  free(sinc);
852  free(colat);
853 
854  return ylat;
855 
856 } /* end gauss2lats() */
