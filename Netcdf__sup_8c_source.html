334 int f_nc_pack(ARG1)
335 {
336  char * pack_to=NULL;
337  float range;
338  int i;
339  if (mode == -1)
340  {
341 //fprintf(stderr,"nc_pack: get arg: %s\n", arg1);
342  pack_to = (char*) malloc((strlen(arg1)+12)*sizeof(char));
343  if (pack_to == NULL) fatal_error("nc_pack: error allocating tmp string","");
344  i = sscanf(arg1,"%g:%g:%s", &nc_valid_min, &nc_valid_max, pack_to);
345  if ( i < 2)
346  {
347  fatal_error("nc_pack: bad value, expect min:max[:byte|short|float(default)], found %s", arg1);
348 // if (sscanf(arg1,"%g:%g:%s", &nc_pack_offset, &nc_pack_scale, pack_to) != 3){
349 // fatal_error("netcdf: bad nc_pack, expect offset:scale:[byte|short], found %s", arg1);
350  }
351  range = (nc_valid_max - nc_valid_min);
352  if ( i == 2 ) strcpy(pack_to,"float");
353 
354  if (strcmp(pack_to,"byte")==0 || strcmp(pack_to,"BYTE")==0 || strcmp(pack_to,"Byte")==0)
355  nc_pack = G2NC_PACK_BYTE;
356  else if (strcmp(pack_to,"short")==0 || strcmp(pack_to,"SHORT")==0 || strcmp(pack_to,"Short")==0)
357  nc_pack = G2NC_PACK_SHORT;
358  else if (strcmp(pack_to,"float")==0 || strcmp(pack_to,"FLOAT")==0 || strcmp(pack_to,"Float")==0)
359  nc_pack = G2NC_PACK_FLOAT; /* will check valid_range */
360  else
361  fatal_error("nc_pack: bad value, expect min:max:byte|short|float, found %s", arg1);
362  if ( nc_pack == G2NC_PACK_FLOAT && fabs(range) < 1e-20 ) fatal_error(
363  "nc_pack: small valid_range specified, expected min:max[:float], min<max, found %s", arg1);
364 
365  /* Center near 0 as signed char or short are used. */
366  if (nc_pack != G2NC_PACK_FLOAT)
367  {
368  nc_pack_offset = (float) ((nc_valid_max+nc_valid_min)*0.5);
369  if ( nc_pack == G2NC_PACK_BYTE )
370  nc_pack_scale = (float) ((range/(G2NC_FILL_VALUE_BYTE - 2))*0.5);
371  else if ( nc_pack == G2NC_PACK_SHORT )
372  nc_pack_scale = (float) ((range/(G2NC_FILL_VALUE_SHORT - 2))*0.5);
373  }
374  free(pack_to);
375 
376 //fprintf(stderr,"nc_pack: nc_pack=%d\n", nc_pack);
377 
378  }
379  return 0;
380 }
