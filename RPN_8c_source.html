1323 int push(int top, unsigned int ndata, int type, float f, float *ff, double *d) {
1324 
1325  unsigned int i;
1326 
1327  if (++top == STACK_SIZE) fatal_error_i("rpn: push: stack overflow %d",top);
1328  if (stack[top] == NULL) {
1329  stack[top] = (float *) malloc(sizeof(float) * (size_t) ndata);
1330  if (stack[top] == NULL) fatal_error("rpn: push: memory allocation","");
1331  }
