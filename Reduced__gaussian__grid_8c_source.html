327 static void interpolate(float *in, int n_in, float *out, int n_out, enum interpol interpolation) {
328  unsigned int i, j, jp1, def_j, def_jp1;
329  double r;
330  if (n_in == n_out) {
331  for (i = 0; i < n_in; i++) {
332  out[i] = in[i];
333  }
334  }
335  else {
336 #ifdef USE_OPENMP
337 #pragma omp parallel for private(i,r,j,jp1, def_j, def_jp1)
338 #endif
339  for (i = 0; i < n_out; i++) {
340  r = i * (double) n_in / (double) (n_out);
341  j = floor(r);
342  r = r - j;
343 
344  jp1 = j + 1;
345  if (j >= n_in) j -= n_in;
346  if (jp1 >= n_in) jp1 -= n_in;
347  def_j = DEFINED_VAL(in[j]);
348  def_jp1 = DEFINED_VAL(in[jp1]);
349 
350  if (r <= 0.0) {
351  out[i] = in[j];
352  }
353  else if (r >= 1.0) {
354  out[i] = in[jp1];
355  }
356  else if (def_j && def_jp1) {
357  if (interpolation == linear || interpolation == linear_extrapolate) { // linear
358  out[i] = (1.0-r)*in[j] + r*in[jp1];
359  }
360  else {
361  out[i] = (r < 0.5) ? in[j] : in[jp1];
362  }
363  }
364  else if (interpolation == linear || interpolation == neighbor) {
365  out[i] = UNDEFINED;
366  }
367  else {
368  out[i] = def_j ? in[j] : in[jp1];
369  }
370  }
371  }
