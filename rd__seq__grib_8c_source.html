455  c3 = getchar_seq_file(input);
456  c4 = getchar_seq_file(input);

459  continue;
460  }

462  unget_seq_file(input, c4);
463  unget_seq_file(input, c3);
464  unget_seq_file(input, c2);
465  unget_seq_file(input, c1);
466  continue;
467  }
