500 unsigned char *mk_bms(float *data, unsigned int *ndata) {
501 
502  int bms_size;
503  unsigned char *bms, *cbits;
504  unsigned int nn, i, start, c, imask, i0;
505 
506  nn = *ndata;
507 
508  /* find first grid point with undefined data */
509  for (i = 0; i < nn; i++) {
510  if (UNDEFINED_VAL(data[i])) break;
511  }
512 
513  if (i == nn) { /* all defined values, no need for bms */
514  bms = (unsigned char *) malloc(6);
515  if (bms == NULL) fatal_error("mk_bms: memory allocation problem","");
516  uint_char(6, bms); // length of section 6
517  bms[4] = 6; // section 6
518  bms[5] = 255; // no bitmap
519  return bms;
520  }
521 
522  /* bms_size = 6 + (nn+7) / 8; */
523  /* nn+7 can overflow */
524  bms_size = 6 + nn/8 + ((nn%8) != 0);
525 
526  bms = (unsigned char *) malloc(bms_size);
527  if (bms == NULL) fatal_error("mk_bms: memory allocation problem","");
528 
529  uint_char(bms_size, bms); // length of section 6
530  bms[4] = 6; // section 6
531  bms[5] = 0; // has bitmap
532 
533  /* bitmap is accessed by bytes, make i0=i/8 bytes of bitmap */
534  cbits = bms + 6;
535  i0 = i >> 3;
536  for (i = 0; i < i0; i++) {
537  *cbits++ = 255;
538  }
539 
540  /* start processing data, skip i0*8 */
541 
542  c = 0;
543  imask = 128;
544  i0 = i0 << 3;
545  start = i0;
546  for (i = i0; i < nn; i++) {
547  if (DEFINED_VAL(data[i])) {
548  c += imask;
549  data[start++] = data[i];
550  }
551  if ((imask >>= 1) == 0) {
552  *cbits++ = c;
553  c = 0;
554  imask = 128;
555  }
556  }
557  if (imask != 128) *cbits = c;
558  *ndata = start;
559  return bms;
560 }

825 void flist2bitstream(float *list, unsigned char *bitstream, unsigned int ndata, int nbits)
826 {
827 
828  int cbits, jbits;
829  unsigned int j, c;
830 
831  if (nbits == 0) {
832  return;
833  }
834  if (nbits < 0) fatal_error_i( "flist2bitstream nbits < 0! nbits = %d", nbits);
835 
836  cbits = 8;
837  c = 0;
838  while (ndata-- > 0) {
839  /* note float -> unsigned int .. truncate */
840  j = (unsigned int) (*list++ + 0.5);
841  jbits = nbits;
842  while (cbits <= jbits) {
843  if (cbits == 8) {
844  jbits -= 8;
845  *bitstream++ = (j >> jbits) & 255;
846  }
847  else {
848  jbits -= cbits;
849  *bitstream++ = (c << cbits) + ((j >> jbits) & mask[cbits]);
850  cbits = 8;
851  c = 0;
852  }
853  }
854  /* now jbits < cbits */
855  if (jbits) {
856  c = (c << jbits) + (j & mask[jbits]);
857  cbits -= jbits;
858  }
859  }
860  if (cbits != 8) *bitstream++ = c << cbits;
861 }
