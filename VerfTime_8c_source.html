152 int verftime(unsigned char **sec, int *year, int *month, int *day, int *hour, int *minute, int *second) {
153 
154  int units, i, j;
155  int dtime;
156  static int error_count = 0;
157  static int warning_count = 0;
158 
159  i = code_table_4_0(sec);
160 
161  // if statistically processed field, verftime is in header
162  j = stat_proc_verf_time(sec, year, month, day, hour, minute, second);
163  if (j == 0) return 0;
164 
165  get_time(sec[1]+12, year, month, day, hour, minute, second);
166 
167  // some products have no forecast time
168 
169  if (code_table_4_4(sec) == -1) return 0;
170 
171  // check the significance of the refernce time
172  i = code_table_1_2(sec);
173 
174  if (i == 2) {
175  // rt=verifying time of forecast
176  // unclear what it means for time averages/accumulations
177  if (warning_count == 0) {
178  fprintf(stderr,"Warning: rt == vt (CodeTable 1.2)\n");
179  warning_count++;
180  }
181  return 0;
182  }
183  if (i == 3) {
184  // rt = observing time
185  return 0;
186  }
187 
188  if (i != 0 && i != 1) {
189  if (error_count == 0) {
190  fprintf(stderr,"verifying time: Table 1.2=%d not supported "
191  " using RT=analysis/start of forecast\n", i);
192  error_count++;
193  }
194  }
195 
196  units = code_table_4_4(sec);
197  dtime = forecast_time_in_units(sec);
198  if (dtime >= 0)
199  return add_time(year, month, day, hour, minute, second, (unsigned int) dtime, units);
200  return sub_dt(year, month, day, hour, minute, second, (unsigned int) (unsigned int) -dtime, units);
201 }
