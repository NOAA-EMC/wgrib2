78 float *ij2p(unsigned int i, unsigned j, int scan_mode, unsigned int nx, unsigned int ny, float *data) {
79 
80  if (scan_mode == -1) return NULL;
81 
82  /* regular grid */
83  if (nx > 0 && ny > 0) {
84  if (i >= nx || j >= ny) return NULL;
85 
86  j = (scan_mode & 64) ? j : ny-1 - j;
87  i = ((scan_mode & 16) && (j % 2 == 1)) ? nx - 1 - i : i;
88  i = (scan_mode & 128) ? nx-1 - i : i;
89 
90  return (scan_mode & 32) ? data + j + i*ny : data + i + nx*j;
